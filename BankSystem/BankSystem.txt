
package BankSystem;

public class BankSystemm {

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        
    
        Customer customer1 = new Customer("omar", new Date(1,5,1999), "falah street");
        Customer customer2 = new Customer("khalid", new Date(1,8,1993), "kahlidiyah");
        Customer customer3 = new Customer("abdullah", new Date(1,3,1995), "khalifa street");
        Customer customer4 = new Customer("waleed", new Date(2,10,2000), "hamdan street");

        BankAccount account1 = new BankAccount("5345", customer1, AccountType.Checking, 800);
        BankAccount account2 = new BankAccount("3323", customer2, AccountType.Saving, 1300);
        BankAccount account3 = new BankAccount("4324", customer3, AccountType.MoneyMarket, 500);
        BankAccount account4 = new BankAccount("5432", customer4, AccountType.Retirement, 8000);

        System.out.println("Bank Account Information:");
        System.out.println(account1);
        System.out.println(account2);
        System.out.println(account3);
        System.out.println(account4);

        System.out.println("\nPerforming deposit operation:");
        System.out.println("Deposit result for account 1: " + (account1.deposit(100) ? "Success" : "Failed"));
        System.out.println("Deposit result for account 2: " + (account2.deposit(100) ? "Success" : "Failed"));
        System.out.println("Deposit result for account 3: " + (account3.deposit(100) ? "Success" : "Failed"));
        System.out.println("Deposit result for account 4: " + (account4.deposit(100) ? "Success" : "Failed"));

        
        System.out.println("\nUpdated Bank Account Information:");
        System.out.println(account1);
        System.out.println(account2);
        System.out.println(account3);
        System.out.println(account4);
    }
}
